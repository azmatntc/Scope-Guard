```python
# change_requests/models.py
from django.db import models
from django.utils import timezone
import uuid
from projects.models import Project

class ChangeRequest(models.Model):
    STATUS_CHOICES = [
        ('DRAFT', 'Draft'),
        ('SENT', 'Sent to Client'),
        ('APPROVED', 'Approved'),
        ('REJECTED', 'Rejected'),
        ('REVISED', 'Revision Requested'),
    ]
    
    id = models.UUIDField(primary_key=True, default=uuid.uuid4, editable=False)
    project = models.ForeignKey(Project, on_delete=models.CASCADE, related_name='change_requests')
    title = models.CharField(max_length=255)
    description = models.TextField()
    estimated_hours = models.DecimalField(max_digits=6, decimal_places=2)
    hourly_rate_cents = models.PositiveIntegerField()  # Snapshot at creation time
    total_cents = models.PositiveIntegerField(editable=False)  # Auto-calculated
    deadline = models.DateField(null=True, blank=True)
    status = models.CharField(max_length=20, choices=STATUS_CHOICES, default='DRAFT')
    approval_token = models.UUIDField(unique=True, default=uuid.uuid4, editable=False, db_index=True)
    approved_at = models.DateTimeField(null=True, blank=True)
    approved_by_ip = models.GenericIPAddressField(null=True, blank=True)
    client_comments = models.TextField(blank=True)
    created_at = models.DateTimeField(auto_now_add=True)
    updated_at = models.DateTimeField(auto_now=True)
    
    class Meta:
        ordering = ['-created_at']
        indexes = [
            models.Index(fields=['project', 'status']),
            models.Index(fields=['approval_token']),
            models.Index(fields=['organization', 'created_at']),  # via project__organization
        ]
    
    def save(self, *args, **kwargs):
        # Auto-calculate total on save
        if self.estimated_hours and self.hourly_rate_cents:
            self.total_cents = int(float(self.estimated_hours) * self.hourly_rate_cents)
        super().save(*args, **kwargs)
    
    @property
    def total_formatted(self):
        return f"${self.total_cents / 100:.2f}"
    
    @property
    def organization(self):
        return self.project.organization
```

#### 2.2 State Machine Logic (Django Signals)
```python
# change_requests/signals.py
from django.db.models.signals import pre_save, post_save
from django.dispatch import receiver
from django.core.mail import send_mail
from django.conf import settings
from .models import ChangeRequest

@receiver(pre_save, sender=ChangeRequest)
def validate_status_transitions(sender, instance, **kwargs):
    """Prevent invalid state transitions"""
    if instance.pk:  # Updating existing
        old_instance = ChangeRequest.objects.get(pk=instance.pk)
        valid_transitions = {
            'DRAFT': ['SENT', 'DRAFT'],
            'SENT': ['APPROVED', 'REJECTED', 'REVISED', 'SENT'],
            'REVISED': ['SENT', 'DRAFT'],
            'APPROVED': ['APPROVED'],  # Locked
            'REJECTED': ['DRAFT', 'REVISED'],
        }
        if instance.status not in valid_transitions.get(old_instance.status, []):
            raise ValueError(f"Invalid transition: {old_instance.status} → {instance.status}")

@receiver(post_save, sender=ChangeRequest)
def send_approval_email(sender, instance, created, **kwargs):
    """Send email when change request is sent to client"""
    if instance.status == 'SENT' and (created or instance.status != instance._state.fields_cache.get('status')):
        approval_url = f"{settings.FRONTEND_URL}/approve/{instance.approval_token}"
        subject = f"Action Required: Approve scope change for {instance.project.name}"
        message = f"""
        Hi {instance.project.client_contact.name},
        
        {instance.project.organization.name} has requested approval for a scope change:
        
        Title: {instance.title}
        Description: {instance.description}
        Estimated Cost: ${instance.total_cents / 100:.2f}
        Deadline: {instance.deadline or 'None'}
        
        Review and approve here: {approval_url}
        
        This link expires in 14 days.
        """
        send_mail(
            subject=subject,
            message=message,
            from_email=settings.DEFAULT_FROM_EMAIL,
            recipient_list=[instance.project.client_contact.email],
            fail_silently=False,
        )
```

#### 2.3 Approval Endpoint (Public, Token-Based)
```python
# change_requests/views.py (add to existing file)
from rest_framework.decorators import api_view, permission_classes
from rest_framework.permissions import AllowAny
from rest_framework.response import Response
from rest_framework import status
from django.utils import timezone
from .models import ChangeRequest

@api_view(['GET', 'PATCH'])
@permission_classes([AllowAny])
def public_approval_view(request, token):
    """Public endpoint for client approval - no auth required"""
    try:
        change_request = ChangeRequest.objects.select_related('project__client_contact').get(
            approval_token=token,
            status='SENT'
        )
    except ChangeRequest.DoesNotExist:
        return Response(
            {'error': {'code': 'NOT_FOUND', 'message': 'Invalid or expired approval link'}},
            status=status.HTTP_404_NOT_FOUND
        )
    
    if request.method == 'GET':
        # Return minimal public data
        return Response({
            'project_name': change_request.project.name,
            'organization_name': change_request.project.organization.name,
            'title': change_request.title,
            'description': change_request.description,
            'total_formatted': change_request.total_formatted,
            'deadline': change_request.deadline,
            'branding': change_request.project.organization.branding
        })
    
    elif request.method == 'PATCH':
        decision = request.data.get('decision')  # 'approve', 'reject', 'revise'
        comments = request.data.get('comments', '')
        
        if decision == 'approve':
            change_request.status = 'APPROVED'
            change_request.approved_at = timezone.now()
            change_request.approved_by_ip = request.META.get('REMOTE_ADDR')
            change_request.client_comments = comments
            change_request.save(update_fields=['status', 'approved_at', 'approved_by_ip', 'client_comments'])
            
            # TODO: Trigger Slack webhook if configured (stub for MVP)
            
            return Response({'status': 'approved', 'message': 'Change request approved successfully'})
        
        elif decision == 'reject':
            change_request.status = 'REJECTED'
            change_request.client_comments = comments
            change_request.save(update_fields=['status', 'client_comments'])
            return Response({'status': 'rejected', 'message': 'Change request rejected'})
        
        elif decision == 'revise':
            change_request.status = 'REVISED'
            change_request.client_comments = comments
            change_request.save(update_fields=['status', 'client_comments'])
            return Response({'status': 'revised', 'message': 'Revision requested - PM will follow up'})
        
        else:
            return Response(
                {'error': {'code': 'INVALID_DECISION', 'message': 'Decision must be approve, reject, or revise'}},
                status=status.HTTP_400_BAD_REQUEST
            )
```

```svelte
  import { onMount } from 'svelte';

  // Props

  // Reactive state using Svelte 5 runes
  let title = $state('');
  let description = $state('');
  let estimatedHours = $state('');
  let deadline = $state('');
  let isSubmitting = $state(false);
  
  // Derived state: auto-calculate total
  let totalFormatted = $derived(() => {
```

```python
# change_requests/views.py (add export endpoint)
@api_view(['GET'])
@permission_classes([permissions.IsAuthenticated])
def export_change_requests_csv(request, project_id):
    """Export approved change requests as QuickBooks-compatible CSV"""
    from django.http import HttpResponse
    import csv
    import io
    
    # Validate project access
    project = get_object_or_404(Project, id=project_id, organization=request.user.organization)
    
    # Get approved change requests
    change_requests = ChangeRequest.objects.filter(
        project=project,
        status='APPROVED'
    ).select_related('project__client_contact').order_by('approved_at')
    
    # Create CSV in memory
    buffer = io.StringIO()
    writer = csv.writer(buffer)
    
    # Header row (QuickBooks Import Format)
    writer.writerow([
        'Date', 'Name', 'Item', 'Description', 'Qty', 'Rate', 'Amount', 
        'Customer:Job', 'Class', 'Memo'
    ])
    
    # Data rows
    for cr in change_requests:
        writer.writerow([
            cr.approved_at.strftime('%m/%d/%Y'),  # Date
            cr.project.client_contact.name,         # Customer name
            'Scope Change',                          # Item/service name
            f"{cr.title}: {cr.description[:50]}...", # Description (truncated)
            float(cr.estimated_hours),               # Quantity (hours)
            f"{cr.hourly_rate_cents / 100:.2f}",    # Rate per hour
            f"{cr.total_cents / 100:.2f}",          # Total amount
            f"{cr.project.client_contact.name}: {cr.project.name}",  # Customer:Job
            '',                                      # Class (optional)
            f"Approved via ScopeGuard - Token: {cr.approval_token}"  # Memo/audit trail
        ])
    
    # Prepare HTTP response
    buffer.seek(0)
    response = HttpResponse(buffer.getvalue(), content_type='text/csv')
    response['Content-Disposition'] = f'attachment; filename="scopeguard_{project.slug}_export_{timezone.now().date()}.csv"'
    return response
```

# SECURITY & PRODUCTION REQUIREMENTS (ENFORCE IN ALL CODE)

## Authentication & Authorization
✅ All PM endpoints: `permission_classes = [IsAuthenticated, IsOrganizationMember]`  
✅ Client approval endpoint: `@permission_classes([AllowAny])` + token validation + rate limiting  
✅ Token security: approval_token = UUID4, expires after 14 days (check in view), single-use for state changes  
✅ Rate limiting: DRF throttling - 100 requests/hour for authenticated users, 5 submissions/hour per approval token  

## Data Integrity & Audit
✅ All financial fields stored in cents (integer) to avoid floating point errors  
✅ `transaction.atomic()` on status changes that trigger side effects (email, Slack)  
✅ Audit middleware: log all CREATE/UPDATE/DELETE on ChangeRequest with user_id, IP, timestamp, old/new values (JSONB)  
✅ Soft delete pattern for critical data: add `is_deleted` + `deleted_at` fields with custom manager filtering  

## Performance Optimization
✅ Database indexes: (organization, status), (approval_token), (project, created_at)  
✅ Query optimization: `select_related('project__client_contact')`, `prefetch_related` for lists  
✅ Pagination: 25 items per page on list endpoints with cursor pagination  
✅ Caching: Cache project summary counts for 2 minutes (django-cacheops or simple in-memory for MVP)
