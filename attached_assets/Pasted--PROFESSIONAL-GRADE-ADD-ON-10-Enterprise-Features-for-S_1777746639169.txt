# integrations/tasks.py (Celery or django-q for async delivery)
def deliver_webhook(webhook_id, event_type, payload):
    subscription = WebhookSubscription.objects.get(id=webhook_id, is_active=True)
    
    # Sign payload for security
    signature = hmac.new(
        subscription.secret.encode(),
        json.dumps(payload, sort_keys=True).encode(),
        hashlib.sha256
    ).hexdigest()
    
    headers = {
        'Content-Type': 'application/json',
        'X-ScopeGuard-Signature': f'sha256={signature}',
        'X-ScopeGuard-Event': event_type,
        'User-Agent': 'ScopeGuard-Webhook/1.0'
    }
    
    try:
        response = requests.post(
            subscription.url,
            json=payload,
            headers=headers,
            timeout=10,
            allow_redirects=False
        )
        # Log delivery
        WebhookDelivery.objects.create(
            subscription=subscription,
            event_type=event_type,
            payload=payload,
            response_status=response.status_code,
            response_body=response.text[:1000],  # Truncate for storage
            delivered_at=timezone.now()
        )
        return response.status_code < 400
    except requests.RequestException as e:
        # Schedule retry with exponential backoff
        schedule_webhook_retry(webhook_id, event_type, payload, retry_count=1)
        return False
```

            WITH change_metrics AS (
                SELECT 
                    DATE_TRUNC('week', created_at) as week,
                    COUNT(*) FILTER (WHERE status = 'APPROVED') as approved_count,
                    COUNT(*) FILTER (WHERE status = 'REJECTED') as rejected_count,

                FROM change_requests_change_request

                GROUP BY week
            )

        """, [org.id, date_from, date_to])

@api_view(['POST'])
@permission_classes([AllowAny])
def sign_document(request, token):
    """Public endpoint for client e-signature"""
    try:
        doc = Document.objects.get(
            signature_hash=token,
            requires_signature=True,
            signature_status='PENDING'
        )
    except Document.DoesNotExist:

    # Capture signature metadata
    signature_data = {
        'signed_at': timezone.now().isoformat(),

        field_changes=[{'field': 'signature_status', 'new_value': 'SIGNED'}],

# core/throttling.py
from rest_framework.throttling import UserRateThrottle
from django.core.cache import cache

class OrganizationRateThrottle(UserRateThrottle):
    """Rate limit by organization, not just user"""
    scope = 'organization'
    
    def allow_request(self, request, view):
        # Get organization from user or token
        org_id = getattr(request.user, 'organization_id', None) or \
                 getattr(request.auth, 'organization_id', None)
        
        if not org_id:
            return super().allow_request(request, view)
        
        # Custom key: org_id + endpoint
        key = f"throttle_org_{org_id}_{view.__class__.__name__}"
        now = time.time()
        window = self.get_rate_limit_window()  # e.g., 3600 seconds
        
        # Get current count
        history = cache.get(key, [])
        history = [t for t in history if t > now - window]
        
        if len(history) >= self.num_requests:
            return False
        
        history.append(now)
        cache.set(key, history, window)
        return True

# In settings.py
REST_FRAMEWORK = {
    'DEFAULT_THROTTLE_CLASSES': [
        'core.throttling.OrganizationRateThrottle',
        'rest_framework.throttling.AnonRateThrottle',
    ],
    'DEFAULT_THROTTLE_RATES': {
        'organization': '1000/hour',  # Per-org limit
        'anon': '100/hour',
        'approval_token': '5/hour',  # Stricter for public endpoints
    }
}

from django.contrib.postgres.search import SearchVector, SearchQuery, SearchRank

def search_change_requests(organization, query_text, filters):
    """Combined full-text + structured filter search"""
    qs = ChangeRequest.objects.select_related('project__client_contact').filter(
        organization=organization
    )
    
    # Full-text search on title + description
    if query_text:
        qs = qs.annotate(
            search=SearchVector('title', 'description', config='english'),
            rank=SearchRank(SearchVector('title', 'description', config='english'), SearchQuery(query_text, config='english'))
        ).filter(search=SearchQuery(query_text, config='english')).order_by('-rank')
    
    # Structured filters
    if filters.get('status'):
        qs = qs.filter(status__in=filters['status'])
    if filters.get('date_from'):
        qs = qs.filter(created_at__gte=filters['date_from'])
    if filters.get('date_to'):
        qs = qs.filter(created_at__lte=filters['date_to'])
    if filters.get('min_amount_cents'):
        qs = qs.filter(total_with_tax_cents__gte=filters['min_amount_cents'])
    if filters.get('client_email'):
        qs = qs.filter(project__client_contact__email__iexact=filters['client_email'])
    
    # Prefetch for list view performance
    return qs.prefetch_related('attachments', 'auditlog_set')
```
