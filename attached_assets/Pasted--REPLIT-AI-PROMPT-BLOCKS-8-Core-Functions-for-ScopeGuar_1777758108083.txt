from decimal import Decimal, ROUND_HALF_UP, InvalidOperation
from typing import Optional

def calculate_scope_total_cents(
    hours: float, 
    rate_cents: int, 
    tax_rate: Optional[Decimal] = None,
    currency_rate: Optional[Decimal] = None  # vs USD base
) -> dict:
    """
    Calculate scope change totals with accounting-grade precision.
    
    Args:
        hours: Estimated hours (float, >0)
        rate_cents: Hourly rate in cents (int, >=0)
        tax_rate: Optional tax percentage as Decimal('8.5') for 8.5%
        currency_rate: Optional exchange rate vs USD as Decimal('0.85') for EUR
    
    Returns:
        dict with integer cent values:
        {
            'subtotal_cents': int,      # hours * rate, rounded
            'tax_cents': int,           # subtotal * tax_rate, rounded  
            'total_cents': int,         # subtotal + tax
            'base_usd_cents': int       # total converted to USD base
        }
    
    Rounding: ROUND_HALF_UP (accounting standard)
    All outputs are integers to prevent float drift.
    """
    # IMPLEMENTATION REQUIREMENTS:
    # 1. Validate inputs: hours > 0, rate_cents >= 0, tax_rate 0-100 if provided
    # 2. Use Decimal for ALL arithmetic to avoid float precision errors
    # 3. Convert float hours to Decimal via str() to prevent binary float issues
    # 4. Apply ROUND_HALF_UP at EACH calculation step (subtotal, tax, conversion)
    # 5. Return ONLY integer cent values (no floats in output)
    # 6. Handle None tax_rate/currency_rate gracefully (skip calculation)
    # 7. Raise ValueError with clear message for invalid inputs
    
    # EDGE CASES TO TEST:
    # - hours=0.25, rate_cents=15000 → subtotal=3750 cents ($37.50)
    # - tax_rate=Decimal('8.375') → compound rounding must match QuickBooks
    # - currency_rate=Decimal('0.0001') → prevent underflow to zero
    # - Very large values: hours=1000, rate_cents=50000 → no overflow

# REPLIT.AI PROMPT: Implement validate_state_transition()

## CONTEXT
ScopeGuard change requests follow a strict workflow: DRAFT → SENT → (APPROVED|REJECTED|REVISED). Invalid transitions must be blocked at the API layer to prevent data corruption and audit failures.

## FUNCTION SPEC
Implement this Python function in `change_requests/utils/workflow.py`:

```python
from enum import Enum, auto
from dataclasses import dataclass
from typing import Set, Optional

class ChangeRequestStatus(Enum):
    DRAFT = 'draft'
    SENT = 'sent' 
    APPROVED = 'approved'
    REJECTED = 'rejected'
    REVISED = 'revised'

@dataclass(frozen=True)
class TransitionRule:
    allowed_targets: Set[ChangeRequestStatus]
    required_permissions: Set[str]  # Django permission codenames
    requires_internal_approval: bool = False

# Define transition rules as module-level constant
TRANSITION_RULES: dict[ChangeRequestStatus, TransitionRule] = {
    ChangeRequestStatus.DRAFT: TransitionRule(
        allowed_targets={ChangeRequestStatus.SENT},
        required_permissions={'change_requests.can_send_to_client'}
    ),
    ChangeRequestStatus.SENT: TransitionRule(
        allowed_targets={
            ChangeRequestStatus.APPROVED,
            ChangeRequestStatus.REJECTED, 
            ChangeRequestStatus.REVISED
        },
        required_permissions=set(),  # Client token auth handles this
        requires_internal_approval=False
    ),
    ChangeRequestStatus.REVISED: TransitionRule(
        allowed_targets={ChangeRequestStatus.SENT, ChangeRequestStatus.DRAFT},
        required_permissions={'change_requests.can_send_to_client'}
    ),
    ChangeRequestStatus.APPROVED: TransitionRule(
        allowed_targets=set(),  # Terminal state
        required_permissions=set()
    ),
    ChangeRequestStatus.REJECTED: TransitionRule(
        allowed_targets={ChangeRequestStatus.DRAFT, ChangeRequestStatus.REVISED},
        required_permissions={'change_requests.can_draft_change'}
    ),
}

@dataclass
class TransitionResult:
    allowed: bool
    error_code: Optional[str] = None
    error_message: Optional[str] = None
    requires_additional_approval: bool = False

def validate_state_transition(
    current_status: ChangeRequestStatus,
    target_status: ChangeRequestStatus,
    user_permissions: Set[str],  # Django permission codenames user has
    is_client_token: bool = False,
    organization_settings: Optional[dict] = None  # For org-specific rules
) -> TransitionResult:
    """
    Validate if a state transition is allowed per workflow rules.
    
    Args:
        current_status: Current CR status enum
        target_status: Desired new status enum
        user_permissions: Set of permission codenames the user possesses
        is_client_token: True if request is from client approval link (bypasses some perms)
        organization_settings: Org-specific overrides (e.g., require internal approval)
    
    Returns:
        TransitionResult with allowed flag and error details if blocked
    """
    # IMPLEMENTATION REQUIREMENTS:
    # 1. Look up rule for current_status in TRANSITION_RULES
    # 2. If target_status not in rule.allowed_targets → block with INVALID_TRANSITION
    # 3. If not is_client_token AND rule.required_permissions not subset of user_permissions → block with INSUFFICIENT_PERMISSIONS
    # 4. If organization_settings.get('require_internal_approval') AND target_status == APPROVED → set requires_additional_approval=True
    # 5. Return TransitionResult with appropriate fields populated
    # 6. Handle unknown status values gracefully (return blocked with UNKNOWN_STATUS)
    
    # ERROR CODES (use for frontend error handling):
    # 'INVALID_TRANSITION', 'INSUFFICIENT_PERMISSIONS', 'UNKNOWN_STATUS', 'ORGANIZATION_POLICY_BLOCKED'
```

# REPLIT.AI PROMPT: Implement build_compliant_queryset()

## CONTEXT
ScopeGuard dashboards must load 50k+ change requests with filters in <500ms. N+1 queries are the #1 performance killer. This function provides a composable, indexed, prefetched QuerySet builder.

## FUNCTION SPEC

from django.db.models import Q, Prefetch, Count, Sum, F, Value
from django.db.models.functions import TruncDate, Coalesce
from django.utils import timezone
from typing import Optional, Dict, Any
from change_requests.models import ChangeRequest, Document
from audit.models import AuditLog

def build_compliant_queryset(
    organization_id: str,
    filters: Optional[Dict[str, Any]] = None,
    user_id: Optional[str] = None,
    include_annotations: bool = True
):
    """
    Build a prefetched, filtered, annotated QuerySet for ChangeRequest.
    
    Args:
        organization_id: UUID string for org scoping (REQUIRED for multi-tenancy)
        filters: Dict of filter params:
            - status: list[str] of status values to include
            - date_from/date_to: date objects for created_at range
            - client_email: str for exact client contact email match
            - min_amount_cents/max_amount_cents: int range for total_with_tax_cents
            - search: str for full-text search on title+description
            - project_id: UUID string to filter by project
        user_id: Optional UUID for permission-based filtering (future-proofing)
        include_annotations: bool to skip heavy annotations for list views
    
    Returns:
        QuerySet ready for serialization with:
        - select_related on FKs (project, client_contact, currency, tax_rate)
        - prefetch_related on reverse FKs (attachments, audit logs limited to 10)
        - Composable filters using indexed fields only
        - Optional annotations: approval_time_hours, attachment_count
        - Ordered by -created_at by default
    """
    # IMPLEMENTATION REQUIREMENTS:
    # 1. Start with base QS: ChangeRequest.objects.filter(organization_id=organization_id)
    # 2. Apply select_related for: project, project__client_contact, currency, tax_rate
    # 3. Apply prefetch_related with limited subqueries:
    #    - attachments: Document.objects.only('id', 'filename', 'file_size_bytes', 'signature_status')
    #    - auditlog_set: AuditLog.objects.order_by('-created_at').only('action', 'created_at')[:10]
    # 4. Apply composable filters (ALL must use indexed fields):
    #    - status__in, created_at__range, project__client_contact__email__iexact
    #    - total_with_tax_cents__range, project_id
    #    - Full-text search using Postgres: to_tsvector + plainto_tsquery with 'english' config
    # 5. If include_annotations=True, add:
    #    - approval_time_hours: Extract(EPOCH FROM (approved_at - created_at))/3600 (null if not approved)
    #    - attachment_count: Count of related documents
    # 6. Default ordering: -created_at
    # 7. Return QS (do NOT execute .all() or .first())
    
    # PERFORMANCE GUARANTEES:
    # - Zero N+1 queries when serializing with ChangeRequestSerializer
    # - Full-text search uses GIN index on tsvector (document index creation in migration notes)
    # - All filter fields have BTREE indexes (list required indexes in output)

## 🔹 BLOCK 4: `jsonb_diff()` — Audit Trail Engine

```markdown
# REPLIT.AI PROMPT: Implement jsonb_diff()

## CONTEXT
ScopeGuard requires immutable, field-level audit trails for compliance (SOC 2) and client dispute resolution. This function generates minimal, human-readable diffs between model states without bloating storage.

## FUNCTION SPEC
Implement this Python function in `audit/utils/diff.py`:

```python
import json
from datetime import datetime, date
from decimal import Decimal
from typing import Any, Optional
from django.db.models import Model
from django.forms.models import model_to_dict

def jsonb_diff(
    old_instance: Optional[Model], 
    new_instance: Model,
    exclude_fields: Optional[list[str]] = None
) -> list[dict]:
    """
    Generate field-level diff between two model instances for audit logging.
    
    Args:
        old_instance: Previous version of model (None if creation)
        new_instance: Current version of model
        exclude_fields: List of field names to ignore (default: system fields)
    
    Returns:
        List of change records:
        [
            {
                'field': 'status',
                'old': 'draft',
                'new': 'sent',
                'field_type': 'CharField',
                'changed_at': '2026-05-03T14:30:00Z'  # ISO format
            },
            ...
        ]
    
    Rules:
    - Skip fields in exclude_fields or starting with '_'
    - Normalize datetime/date to ISO strings for comparison
    - Handle Decimal → float conversion for JSON serialization
    - Skip unchanged fields (old == new after normalization)
    - Include field_type from model._meta for frontend rendering hints
    """
    # IMPLEMENTATION REQUIREMENTS:
    # 1. Default exclude_fields: ['id', 'created_at', 'updated_at', 'organization_id', '_state']
    # 2. Convert instances to dicts using model_to_dict with exclude parameter
    # 3. For each field in new_dict:
    #    a. Skip if in exclude_fields or starts with '_'
    #    b. Get old_value (None if old_instance is None or field not present)
    #    c. Normalize values for comparison:
    #       - datetime/date → .isoformat()
    #       - Decimal → float (safe for audit display, not for calc)
    #       - UUID → str
    #    d. If normalized old != normalized new, add to changes list
    # 4. For each change, include:
    #    - field name
    #    - old/new values (JSON-serializable)
    #    - field_type from new_instance._meta.get_field(field).get_internal_type()
    #    - changed_at timestamp (use timezone.now().isoformat())
    # 5. Return sorted list by field name for deterministic output
    
    # PERFORMANCE: O(n) where n = number of model fields (typically <50)
    # STORAGE: Only store changed fields, not full snapshots
```

## INTEGRATION WITH AUDIT LOGGING
```python
# In audit/signals.py
from audit.utils.diff import jsonb_diff

@receiver(post_save, sender=ChangeRequest)
def audit_change_request(sender, instance, created, **kwargs):
    if created:
        # Creation: log all fields as "new"
        changes = [{'field': k, 'old': None, 'new': v, 'field_type': 'unknown'} 
                  for k, v in model_to_dict(instance).items()]
    else:
        # Update: compute diff
        old_instance = sender.objects.get(pk=instance.pk)
        changes = jsonb_diff(old_instance, instance)
    
    if changes:  # Only log if something changed
        AuditLog.objects.create(
            organization=instance.organization,
            action='UPDATE' if not created else 'CREATE',
            content_object=instance,
            field_changes=changes,  # JSONB field
            request_metadata=capture_request_metadata()
        )
```

## CONTEXT

// src/lib/utils/optimistic.ts
import { toast } from '$lib/stores/ui.store';
import type { ChangeRequest } from '$lib/types';

export type OptimisticOptions<T> = {
  // The Svelte store holding the list of items (must have update/set methods)
  store: {
    update: (updater: (items: T[]) => T[]) => void;
    set: (items: T[]) => void;
  };
  
  // Async function that performs the actual API mutation
  mutationFn: (payload: Record<string, unknown>) => Promise<T>;
  
  // The payload to send to the API
  payload: Record<string, unknown>;
  
  // Optional: Custom ID generator for optimistic items (default: crypto.randomUUID)
  generateTempId?: () => string;
  
  // Optional: Custom rollback logic if default replacement isn't sufficient
  onRollback?: (tempItem: T, error: unknown) => void;
  
  // Optional: Custom success handler beyond store update
  onSuccess?: (realItem: T) => void;
};

/**
 * Perform an optimistic UI update with automatic rollback on failure.
 * 
 * Workflow:
 * 1. Create optimistic item with temp ID + _optimistic flag
 * 2. Immediately update store to show item in UI
 * 3. Execute mutationFn in background
 * 4. On success: replace optimistic item with real server response
 * 5. On failure: remove optimistic item + show error toast + call onRollback
 * 
 * @returns Promise that resolves with server response or rejects with error
 */
export async function optimisticMutation<T extends { id: string }>(
  options: OptimisticOptions<T>
): Promise<T> {
  const {
    store,
    mutationFn,
    payload,
    generateTempId = () => crypto.randomUUID(),
    onRollback,
    onSuccess
  } = options;
  
  // Create optimistic item with metadata
  const tempId = generateTempId();
  const optimisticItem: T = {
    ...payload,
    id: tempId,
    _optimistic: true,
    _createdAt: Date.now(),
    status: 'pending' // Custom status for UI indication
  } as T;
  
  // 1. Optimistic update: add to store immediately
  store.update(items => [...items, optimisticItem]);
  
  try {
    // 2. Execute real mutation
    const response = await mutationFn(payload);
    
    // 3. Replace optimistic item with real server data
    store.update(items => 
      items.map(item => 
        (item as any)._optimistic && (item as any).id === tempId 
          ? { ...response, _optimistic: false } 
          : item
      )
    );
    
    // 4. Optional success callback
    onSuccess?.(response);
    
    return response;
  } catch (error: unknown) {
    // 5. Rollback: remove optimistic item
    store.update(items => 
      items.filter(item => 
        !((item as any)._optimistic && (item as any).id === tempId)
      )
    );
    
    // 6. Custom rollback logic if provided
    onRollback?.(optimisticItem, error);
    
    // 7. User feedback
    const errorMessage = (error as any)?.response?.data?.error?.message 
      || 'Action failed. Changes reverted.';
    toast.error(errorMessage);
    
    // Re-throw for caller handling if needed
    throw error;
  }
}

# REPLIT.AI PROMPT: Implement useDebouncedFilter() for Svelte 5

## CONTEXT
ScopeGuard dashboards have complex filters (status, date range, amount, search). Typing should not spam the API. Filters should sync to URL for bookmarking/sharing. Changes should apply after user pauses typing.

## FUNCTION SPEC

```typescript
// src/lib/utils/filters.ts
import { page } from '$app/state';
import { goto } from '$app/navigation';
import { browser } from '$app/environment';

export type FilterConfig = {
  // Debounce delay in milliseconds (default: 300)
  delay?: number;
  
  // Sync filter state to URL query params (default: true)
  urlSync?: boolean;
  
  // Custom serializer for complex filter values (default: JSON.stringify for arrays/objects)
  serializeValue?: (key: string, value: unknown) => string | null;
  
  // Custom deserializer for URL params (default: parse JSON for arrays/objects)
  deserializeValue?: (key: string, value: string) => unknown;
  
  // Fields to exclude from URL sync (e.g., ephemeral UI state)
  excludeFromUrl?: string[];
};

export type FilterActions = {
  // Apply new filters with debouncing
  apply: (newFilters: Record<string, unknown>) => void;
  
  // Immediately apply filters without debounce (for button clicks)
  applyNow: (newFilters: Record<string, unknown>) => void;
  
  // Reset to default filters
  reset: () => void;
  
  // Get current filter state
  get: () => Record<string, unknown>;
  
  // Subscribe to filter changes (for reactive components)
  subscribe: (callback: (filters: Record<string, unknown>) => void) => () => void;
};

/**
 * Create a debounced, URL-synced filter manager for Svelte components.
 * 
 * Features:
 * - Debounce rapid changes to prevent API spam
 * - Sync to URL query params for bookmarkable views
 * - Parse complex values (arrays, dates) to/from URL strings
 * - Reactive subscription for UI updates
 * - Browser-only execution (SSR safe)
 * 
 * @param initialFilters - Starting filter state
 * @param config - Optional configuration for debounce, serialization, etc.
 * @returns FilterActions object with apply/reset/get/subscribe methods
 */
export function useDebouncedFilter(
  initialFilters: Record<string, unknown> = {},
  config: FilterConfig = {}
): FilterActions {
  const {
    delay = 300,
    urlSync = true,
    serializeValue = defaultSerialize,
    deserializeValue = defaultDeserialize,
    excludeFromUrl = []
  } = config;
  
  // Private state
  let currentFilters = { ...initialFilters };
  let timeout: ReturnType<typeof setTimeout> | null = null;
  let subscribers: Array<(filters: Record<string, unknown>) => void> = [];
  
  // Notify subscribers of changes
  const notify = () => {
    subscribers.forEach(cb => cb({ ...currentFilters }));
  };
  
  // Update URL query params (browser only)
  const syncToUrl = (filters: Record<string, unknown>) => {
    if (!browser || !urlSync) return;
    
    const params = new URLSearchParams(page.url.searchParams);
    
    // Clear existing filter params (prefix with 'f_' to avoid collisions)
    Array.from(params.keys()).forEach(key => {
      if (key.startsWith('f_')) params.delete(key);
    });
    
    // Add new params
    Object.entries(filters).forEach(([key, value]) => {
      if (excludeFromUrl.includes(key)) return;
      
      const serialized = serializeValue(key, value);
      if (serialized !== null && serialized !== undefined) {
        params.set(`f_${key}`, serialized);
      }
    });
    
    // Update URL without page reload
    goto(`?${params.toString()}`, { replaceState: true, noScroll: true });
  };
  
  // Parse filters from URL on init (browser only)
  const initFromUrl = () => {
    if (!browser || !urlSync) return initialFilters;
    
    const parsed: Record<string, unknown> = {};
    
    page.url.searchParams.forEach((value, key) => {
      if (key.startsWith('f_')) {
        const filterKey = key.slice(2); // Remove 'f_' prefix
        if (!excludeFromUrl.includes(filterKey)) {
          parsed[filterKey] = deserializeValue(filterKey, value);
        }
      }
    });
    
    return { ...initialFilters, ...parsed };
  };
  
  // Initialize from URL if in browser
  if (browser && urlSync) {
    currentFilters = initFromUrl();
  }
  
  // Debounced apply
  const apply = (newFilters: Record<string, unknown>) => {
    // Merge with current filters
    currentFilters = { ...currentFilters, ...newFilters };
    
    // Clear pending timeout
    if (timeout) clearTimeout(timeout);
    
    // Notify subscribers immediately for UI responsiveness
    notify();
    
    // Debounce URL sync + API trigger
    timeout = setTimeout(() => {
      syncToUrl(currentFilters);
      timeout = null;
    }, delay);
  };
  
  // Immediate apply (no debounce)
  const applyNow = (newFilters: Record<string, unknown>) => {
    currentFilters = { ...currentFilters, ...newFilters };
    notify();
    syncToUrl(currentFilters);
  };
  
  // Reset to initial state
  const reset = () => {
    currentFilters = { ...initialFilters };
    notify();
    syncToUrl(currentFilters);
  };
  
  // Get current filters
  const get = () => ({ ...currentFilters });
  
  // Subscribe to changes
  const subscribe = (callback: (filters: Record<string, unknown>) => void) => {
    subscribers.push(callback);
    // Immediately notify with current state
    callback({ ...currentFilters });
    // Return unsubscribe function
    return () => {
      subscribers = subscribers.filter(cb => cb !== callback);
    };
  };
  
  return { apply, applyNow, reset, get, subscribe };
}

// Default serializers/deserializers
function defaultSerialize(key: string, value: unknown): string | null {
  if (value === null || value === undefined || value === '') return null;
  if (Array.isArray(value)) return value.join(',');
  if (value instanceof Date) return value.toISOString().split('T')[0]; // YYYY-MM-DD
  if (typeof value === 'object') return JSON.stringify(value);
  return String(value);
}

function defaultDeserialize(key: string, value: string): unknown {
  // Try parsing as array first (comma-separated)
  if (value.includes(',')) {
    return value.split(',').map(v => v.trim());
  }
  // Try parsing as date
  if (/^\d{4}-\d{2}-\d{2}$/.test(value)) {
    return new Date(value);
  }
  // Try parsing as JSON object
  if (value.startsWith('{') || value.startsWith('[')) {
    try {
      return JSON.parse(value);
    } catch {
      return value; // Fallback to string
    }
  }
  // Boolean parsing
  if (value === 'true') return true;
  if (value === 'false') return false;
  // Number parsing
  if (!isNaN(Number(value)) && value.trim() !== '') {
    return Number(value);
  }
  return value;
}

## 🔹 BLOCK 7: `atomic_with_audit()` — Transaction-Safe Mutations

```markdown
# REPLIT.AI PROMPT: Implement atomic_with_audit() decorator

## CONTEXT
ScopeGuard financial data must never be partially written. If a change request update succeeds but the audit log fails, we have compliance risk. This decorator wraps DB mutations + audit logging in a single atomic transaction.

## FUNCTION SPEC
Implement this Python decorator in `audit/utils/transaction.py`:

```python
from functools import wraps
from typing import Callable, TypeVar, ParamSpec, Optional
from django.db import transaction
from django.utils import timezone
from django.forms.models import model_to_dict
from audit.models import AuditLog
from audit.utils.diff import jsonb_diff

P = ParamSpec('P')
R = TypeVar('R')

def atomic_with_audit(
    action: str,
    model_instance_arg: str = 'self',  # Name of arg containing the model instance
    metadata_extractor: Optional[Callable[[P], dict]] = None
) -> Callable[[Callable[P, R]], Callable[P, R]]:
    """
    Decorator to wrap model mutations with atomic transaction + audit logging.
    
    Usage:
        @atomic_with_audit(action='UPDATE', model_instance_arg='change_request')
        def update_change_request(change_request: ChangeRequest, **updates):
            for k, v in updates.items():
                setattr(change_request, k, v)
            change_request.save()
    
    Guarantees:
    - If DB save fails, audit log is never written (atomic rollback)
    - If audit log write fails, DB save is rolled back
    - Audit entry captures field-level changes via jsonb_diff
    - Works with both instance methods and standalone functions
    
    Args:
        action: Audit action string ('CREATE', 'UPDATE', 'DELETE', 'APPROVE', etc.)
        model_instance_arg: Name of the function argument that holds the model instance
        metadata_extractor: Optional function to extract request metadata from function args
    
    Returns:
        Decorated function that executes atomically with audit logging
    """
    def decorator(func: Callable[P, R]) -> Callable[P, R]:
        @wraps(func)
        @transaction.atomic
        def wrapper(*args: P.args, **kwargs: P.kwargs) -> R:
            # Extract model instance from args/kwargs
            model_instance = _extract_model_instance(args, kwargs, model_instance_arg)
            organization = _extract_organization(model_instance)
            
            # Capture pre-state for diff
            old_data = model_to_dict(model_instance) if model_instance.pk else None
            
            # Execute the wrapped function
            result = func(*args, **kwargs)
            
            # Refresh instance to get post-save state (handles DB defaults, triggers)
            if model_instance.pk:
                model_instance.refresh_from_db()
            
            # Capture post-state and compute diff
            new_data = model_to_dict(model_instance)
            field_changes = jsonb_diff(old_data, new_data)
            
            # Only create audit log if something changed or it's a creation
            if field_changes or old_data is None:
                # Extract request metadata if extractor provided
                request_metadata = {}
                if metadata_extractor:
                    try:
                        request_metadata = metadata_extractor(*args, **kwargs)
                    except Exception:
                        # Never let metadata extraction break the transaction
                        request_metadata = {'extraction_error': 'Failed to extract metadata'}
                
                # Create audit log entry
                AuditLog.objects.create(
                    organization=organization,
                    action=action,
                    content_object=model_instance,
                    field_changes=field_changes,
                    request_metadata={
                        'timestamp': timezone.now().isoformat(),
                        **request_metadata
                    }
                )
            
            return result
        return wrapper
    return decorator

def _extract_model_instance(args: tuple, kwargs: dict, arg_name: str):
    """Helper to extract model instance from function arguments"""
    if arg_name in kwargs:
        return kwargs[arg_name]
    # Try positional args (naive: assumes order matches function signature)
    # Production: use inspect.signature for robustness
    for arg in args:
        if hasattr(arg, '_meta') and hasattr(arg, 'pk'):  # Django model check
            return arg
    raise ValueError(f"Could not extract model instance from arg '{arg_name}'")

def _extract_organization(model_instance) -> Optional['Organization']:
    """Extract organization from model instance via common patterns"""
    if hasattr(model_instance, 'organization'):
        return model_instance.organization
    if hasattr(model_instance, 'project') and hasattr(model_instance.project, 'organization'):
        return model_instance.project.organization
    return None
```

            user_permissions=set(request.user.get_all_permissions()),
            is_client_token=False
        )
        if not transition_result.allowed:
            raise PermissionDenied(transition_result.error_message)
        
        # Atomic update + audit
        change_request.status = 'approved'
        change_request.approved_at = timezone.now()
        change_request.approved_by_ip = request.META.get('REMOTE_ADDR')
        change_request.save(update_fields=['status', 'approved_at', 'approved_by_ip'])
        
        return Response({'status': 'approved'})

```markdown
# REPLIT.AI PROMPT: Implement webhook signature functions

## CONTEXT
ScopeGuard sends webhooks to Zapier, Make, Slack, and accounting tools. These integrations must be cryptographically signed to prevent spoofing, replay attacks, and tampering. Implementation must match Stripe/GitHub standards.

## FUNCTION SPEC

```python
import hmac
import hashlib
import time
import json
from typing import Union, Optional
from django.conf import settings

def sign_webhook_payload(
    payload: Union[dict, str], 
    secret: str, 
    timestamp: Optional[int] = None
) -> str:
    """
    Sign a webhook payload using HMAC-SHA256 with timestamp.
    
    Format: "t={timestamp},v1={hex_signature}"
    Matches Stripe signature format for compatibility.
    
    Args:
        payload: Dict to sign (will be JSON-serialized) or pre-serialized JSON string
        secret: Webhook signing secret (store in env vars, never in code)
        timestamp: Unix timestamp (default: current time)
    
    Returns:
        Signature string: "t=1714752000,v1=abc123..."
    """
    ts = timestamp or int(time.time())
    
    # Ensure payload is JSON string with sorted keys for deterministic signing
    if isinstance(payload, dict):
        payload_str = json.dumps(payload, sort_keys=True, separators=(',', ':'))
    else:
        payload_str = payload
    
    # Create signed message: timestamp.payload
    signed_payload = f"{ts}.{payload_str}"
    
    # Compute HMAC-SHA256
    signature = hmac.new(
        key=secret.encode('utf-8'),
        msg=signed_payload.encode('utf-8'),
        digestmod=hashlib.sha256
    ).hexdigest()
    
    return f"t={ts},v1={signature}"

def verify_webhook_signature(
    payload: Union[dict, str],
    signature: str,
    secret: str,
    tolerance_seconds: int = 300  # 5 minutes default
) -> bool:
    """
    Verify a webhook signature with timestamp tolerance.
    
    Security checks:
    1. Signature matches HMAC-SHA256 of timestamp.payload
    2. Timestamp within tolerance window (prevent replay attacks)
    3. Constant-time comparison to prevent timing attacks
    
    Args:
        payload: Original payload dict or JSON string
        signature: Signature string from sign_webhook_payload
        secret: Same signing secret used to create signature
        tolerance_seconds: Max age of timestamp to accept (default: 300)
    
    Returns:
        True if signature is valid and timestamp is fresh, False otherwise
    """
    try:
        # Parse signature header: "t=123,v1=abc"
        parts = dict(part.split('=', 1) for part in signature.split(','))
        if 't' not in parts or 'v1' not in parts:
            return False
        
        timestamp = int(parts['t'])
        expected_signature = parts['v1']
        
        # Check timestamp freshness (prevent replay attacks)
        now = int(time.time())
        if abs(now - timestamp) > tolerance_seconds:
            return False
        
        # Re-compute expected signature
        expected_full = sign_webhook_payload(payload, secret, timestamp)
        expected_sig = expected_full.split(',', 1)[1].split('=', 1)[1]
        
        # Constant-time comparison to prevent timing attacks
        return hmac.compare_digest(expected_sig, expected_signature)
        
    except (ValueError, KeyError, TypeError):
        # Any parsing error = invalid signature
        return False
```
